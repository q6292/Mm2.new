local DragHandle = Instance.new("TextButton")

DragHandle.BackgroundTransparency = 1
DragHandle.Text = ""
DragHandle.Parent = TitleBar

local dragging = false
local dragStart, startPos

DragHandle.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = Main.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = input.Position - dragStart
        Main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)

-- Spawn Function
local canSpawn = true

local function spawnWeapon(weaponName)
    local amount = math.random(1, 100)
    
    local success = pcall(function()
        if not PlayerWeapons.Owned[weaponName] then
            PlayerWeapons.Owned[weaponName] = amount
        else
            PlayerWeapons.Owned[weaponName] = PlayerWeapons.Owned[weaponName] + amount
        end
    end)
    
    if not success then
        warn("Failed to add weapon: " .. weaponName)
        return false, 0
    end
    
    pcall(function()
        if LocalPlayer.Character then
            LocalPlayer.Character:BreakJoints()
        end
    end)
    
    return true, amount
end

-- Click Handler
SpawnBtn.MouseButton1Click:Connect(function()
    if not canSpawn then return end
    canSpawn = false
    SpawnBtn.Interactable = false
    SpawnBtn.Text = "..."
    
    -- Filter valid weapons
    local valid = {}
    for _, wName in ipairs(allWeapons) do
        if ItemDatabase[wName] then
            table.insert(valid, wName)
        end
    end
    
    if #valid == 0 then
        WeaponLabel.Text = "No weapons found!"
        SpawnBtn.Text = "SPAWN"
        SpawnBtn.Interactable = true
        canSpawn = true
        return
    end
    
    -- Rolling animation
    for i = 1, 12 do
        local randomWep = valid[math.random(1, #valid)]
        WeaponLabel.Text = randomWep
        IconLabel.TextColor3 = Color3.fromRGB(255, 200, 50)
        task.wait(0.1 + (i * 0.01))
    end
    
    -- Pick final weapon
    local chosen = valid[math.random(1, #valid)]
    local success, amount = spawnWeapon(chosen)
    
    if success then
        IconLabel.Text = "✓"
        IconLabel.TextColor3 = Color3.fromRGB(100, 255, 100)
        WeaponLabel.Text = chosen
        WeaponLabel.TextColor3 = Color3.fromRGB(100, 255, 100)
        AmountLabel.Text = amount .. "x spawned!"
        AmountLabel.TextColor3 = Color3.fromRGB(100, 255, 100)
    else
        IconLabel.Text = "✗"
        IconLabel.TextColor3 = Color3.fromRGB(255, 100, 100)
        WeaponLabel.Text = "Error!"
        WeaponLabel.TextColor3 = Color3.fromRGB(255, 100, 100)
        AmountLabel.Text = "Try again"
    end
    
    -- Reset
    task.wait(2)
    IconLabel.Text = "?"
    IconLabel.TextColor3 = Color3.fromRGB(100, 170, 255)
    WeaponLabel.Text = "Ready"
    WeaponLabel.TextColor3 = Color3.fromRGB(140, 160, 200)
    AmountLabel.Text = ""
    SpawnBtn.Text = "SPAWN"
    SpawnBtn.Interactable = true
    canSpawn = true
end)
