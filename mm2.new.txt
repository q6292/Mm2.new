-- Try multiple module paths
local ItemDatabase = nil
local PlayerData = nil

-- Try every possible path
local paths = {
    {ReplicatedStorage, "Database", "Sync", "Item"},
    {ReplicatedStorage, "Database/Sync/Item"},
}

for _, path in ipairs(paths) do
    local success, result = pcall(function()
        if #path == 4 then
            return require(path[1][path[2]][path[3]][path[4]])
        else
            local parts = path[2]:split("/")
            local obj = path[1]
            for _, part in ipairs(parts) do
                obj = obj[part]
            end
            return require(obj)
        end
    end)
    if success and result then
        ItemDatabase = result
        print("Found ItemDatabase!")
        break
    end
end

-- Try PlayerData paths
local pdPaths = {
    {ReplicatedStorage, "Modules", "ProfileData"},
}

for _, path in ipairs(pdPaths) do
    local success, result = pcall(function()
        return require(path[1][path[2]][path[3]])
    end)
    if success and result then
        PlayerData = result
        print("Found PlayerData!")
        break
    end
end

-- Last resort - search everything
if not ItemDatabase then
    for _, child in ipairs(ReplicatedStorage:GetDescendants()) do
        if child.Name == "Item" and child:IsA("ModuleScript") then
            pcall(function()
                ItemDatabase = require(child)
                print("Found ItemDatabase via search!")
            end)
            if ItemDatabase then break end
        end
    end
end

if not PlayerData then
    for _, child in ipairs(ReplicatedStorage:GetDescendants()) do
        if child.Name == "ProfileData" and child:IsA("ModuleScript") then
            pcall(function()
                PlayerData = require(child)
                print("Found PlayerData via search!")
            end)
            if PlayerData then break end
        end
    end
end

if not ItemDatabase or not PlayerData then
    warn("Could not find MM2 modules after searching. Is this MM2?")
end

    PlayerWeapons = PlayerData.Weapons
end

-- Drag
local DragHandle = Instance.new("TextButton")
DragHandle.Size = UDim2.new(1, -26, 0, 30)
DragHandle.BackgroundTransparency = 1
DragHandle.Text = ""
DragHandle.Parent = TitleBar

local dragging = false
local dragStart, startPos

DragHandle.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = Main.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = input.Position - dragStart
        Main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)

-- Hover

local canSpawn = true

SpawnBtn.MouseButton1Click:Connect(function()
    if not canSpawn then return end
    canSpawn = false
    SpawnBtn.Interactable = false
    SpawnBtn.Text = "..."
    
    if not ItemDatabase then
        WeaponLabel.Text = "Not in MM2!"
        WeaponLabel.TextColor3 = Color3.fromRGB(255, 100, 100)
        AmountLabel.Text = "Join MM2 game"
        task.wait(2)
        WeaponLabel.Text = "Ready"
        WeaponLabel.TextColor3 = Color3.fromRGB(140, 160, 200)
        AmountLabel.Text = ""
        SpawnBtn.Text = "SPAWN"
        SpawnBtn.Interactable = true
        canSpawn = true
        return
    end
    
    local valid = {}
    for _, wName in ipairs(allWeapons) do
        pcall(function()
            if ItemDatabase[wName] then
                table.insert(valid, wName)
            end
        end)
    end
    
    if #valid == 0 then
        WeaponLabel.Text = "Error!"
        WeaponLabel.TextColor3 = Color3.fromRGB(255, 100, 100)
        AmountLabel.Text = "No valid weapons"
        task.wait(2)
        WeaponLabel.Text = "Ready"
        WeaponLabel.TextColor3 = Color3.fromRGB(140, 160, 200)
        AmountLabel.Text = ""
        SpawnBtn.Text = "SPAWN"
        SpawnBtn.Interactable = true
        canSpawn = true
        return
    end
    
    -- Rolling animation
    for i = 1, 12 do
        local rw = valid[math.random(1, #valid)]
        WeaponLabel.Text = rw
        WeaponLabel.TextColor3 = Color3.fromRGB(255, 200, 50)
        IconLabel.TextColor3 = Color3.fromRGB(255, 200, 50)
        task.wait(0.08 + (i * 0.005))
    end
    
    local chosen = valid[math.random(1, #valid)]
    local amount = math.random(1, 100)
    
    local success = pcall(function()
        if not PlayerWeapons or not PlayerWeapons.Owned then return end
        if not PlayerWeapons.Owned[chosen] then
            PlayerWeapons.Owned[chosen] = amount
        else
            PlayerWeapons.Owned[chosen] = PlayerWeapons.Owned[chosen] + amount
        end
    end)
    
    if success and PlayerWeapons and PlayerWeapons.Owned then
        pcall(function()
            if LocalPlayer.Character then
                LocalPlayer.Character:BreakJoints()
            end
        end)
        
        IconLabel.Text = "✓"
        IconLabel.TextColor3 = Color3.fromRGB(100, 255, 100)
        WeaponLabel.Text = chosen
        WeaponLabel.TextColor3 = Color3.fromRGB(100, 255, 100)
        AmountLabel.Text = amount .. "x spawned!"
        AmountLabel.TextColor3 = Color3.fromRGB(100, 255, 100)
    else
        IconLabel.Text = "✗"
        IconLabel.TextColor3 = Color3.fromRGB(255, 100, 100)
        WeaponLabel.Text = "Failed!"
        WeaponLabel.TextColor3 = Color3.fromRGB(255, 100, 100)
        AmountLabel.Text = "Try rejoining"
    end
    
    task.wait(2.5)
    IconLabel.Text = "?"
    IconLabel.TextColor3 = Color3.fromRGB(100, 170, 255)
    WeaponLabel.Text = "Ready"
    WeaponLabel.TextColor3 = Color3.fromRGB(140, 160, 200)
    AmountLabel.Text = ""
    SpawnBtn.Text = "SPAWN"
    SpawnBtn.Interactable = true
    canSpawn = true
end)
