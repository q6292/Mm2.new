local function resetSlots() for _, slot in ipairs(weaponSlots) do slot.frame.BackgroundColor3 = Color3.fromRGB(22, 24, 36) slot.name.TextColor3 = Color3.fromRGB(200, 200, 200) end end

local isOpening = false

local highlightEnd = tick() + 2.5 local lastHighlight = tick() local speed = 0.08
local hc; hc = RunService.Heartbeat:Connect(function()
if tick() > highlightEnd then return end
if tick() - lastHighlight >= speed then resetSlots()
local rs = weaponSlots[math.random(1, #weaponSlots)]
rs.frame.BackgroundColor3 = rs.weapon.color rs.name.TextColor3 = Color3.fromRGB(255, 255, 255)
ScrollFrame.CanvasPosition = Vector2.new(0, math.max(0, (rs.frame.LayoutOrder - 2) * 36))
lastHighlight = tick() speed = math.max(0.03, speed * 0.97)
end end)
task.wait(2.5) if hc then hc:Disconnect() end
local totalChance = 0 for _, w in ipairs(validWeapons) do totalChance = totalChance + w.chance end
local roll = math.random() * totalChance local cumulative = 0 local winner = validWeapons[1]
for _, w in ipairs(validWeapons) do cumulative = cumulative + w.chance if roll <= cumulative then winner = w break end end
local amount = math.random(1, 100)
local winnerSlot = nil for _, slot in ipairs(weaponSlots) do if slot.weapon == winner then winnerSlot = slot break end end
if winnerSlot then
local targetPos = (winnerSlot.frame.LayoutOrder - 2) * 36 local currentPos = ScrollFrame.CanvasPosition.Y local startTime = tick()
local sc; sc = RunService.Heartbeat:Connect(function()
local elapsed = tick() - startTime
if elapsed >= 0.8 then ScrollFrame.CanvasPosition = Vector2.new(0, targetPos) if sc then sc:Disconnect() end return end
local t = elapsed / 0.8 t = 1 - (1 - t) * (1 - t)
ScrollFrame.CanvasPosition = Vector2.new(0, currentPos + (targetPos - currentPos) * t)
end)
task.wait(0.8) if sc then sc:Disconnect() end
end
resetSlots()
if winnerSlot then
for _, slot in ipairs(weaponSlots) do slot.frame.BackgroundColor3 = Color3.fromRGB(255, 255, 255) end
task.wait(0.15) resetSlots()
winnerSlot.frame.BackgroundColor3 = winner.color winnerSlot.name.TextColor3 = Color3.fromRGB(255, 255, 255)
end
ResultLabel.Text = winner.name .. " · " .. amount .. "x" ResultLabel.TextColor3 = winner.color
ResultRarity.Text = winner.rarity .. " | " .. winner.chance .. "% chance" ResultRarity.TextColor3 = winner.color
pcall(function() if not PlayerWeapons.Owned[winner.name] then PlayerWeapons.Owned[winner.name] = amount else PlayerWeapons.Owned[winner.name] = PlayerWeapons.Owned[winner.name] + amount end end)
pcall(function() if LocalPlayer.Character then LocalPlayer.Character:BreakJoints() end end)
task.wait(3) resetSlots() OpenBtn.Text = "OPEN CRATE" OpenBtn.Interactable = true isOpening = false
end)
